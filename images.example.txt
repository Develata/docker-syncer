# 每行一个镜像；空行与整行注释会被忽略。
# 仅支持 --platform 和 --target，可放在镜像前后，也可使用 =。
# --target 必须是无斜杠的小写 repository:tag。

# 默认平台来自 syncer.json（当前为 linux/amd64）
nginx:alpine

# 指定单个平台；linux/arm64 会规范化为 linux/arm64/v8
mysql:8.0 --platform linux/arm64

# 32 位 ARM 必须明确 v6 或 v7
--platform=linux/arm/v7 arm32v7/alpine:latest

# 保留稳定的旧目标名称
bitnami/redis:7.2 --target bitnami_redis:7.2

# 完整复制 manifest list/index，与 runner 架构无关
registry.k8s.io/pause:3.10 --platform all
